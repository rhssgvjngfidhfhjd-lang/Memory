第一步：构建记忆库（a → b → c 顺序跑）

cd /data1/haozhen/Visual_Primitives/Offline/Offline

# mode a: text only
PYTHONPATH=. python3 -m simple_memory_pipeline.build_memgallery_memories \
  --mode a --all-datasets \
  --executor-model Qwen/Qwen3-VL-4B-Instruct \
  --executor-base-url http://127.0.0.1:8000/v1 \
  --executor-api-key EMPTY \
  --device cuda:0 \
  > artifacts/simple_memory_pipeline/logs/build_a.log 2>&1

# mode b: text + caption
